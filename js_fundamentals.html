01 Introduction to JavaScript

JavaScript (JS) is a lightweight, interpreted, single-threaded programming language that runs in browsers and on servers (Node.js). It is the only language natively understood by web browsers, making it the backbone of interactive web pages.

FeatureDetails
Created byBrendan Eich (Netscape, 1995) — in 10 days
StandardECMAScript (ECMA-262) — ES6 (2015) was a major milestone
ParadigmMulti-paradigm: imperative, OOP, functional
TypingDynamically typed, weakly typed
ExecutionBrowser engine (V8, SpiderMonkey) or Node.js
File extension.js — linked with <script src="app.js">
Running JavaScript — three ways
hello.js
// 1. Inline inside HTML

// 2. External file linked in HTML

// 3. Node.js terminal

console.log("Hello, World!");    // prints to console
alert("Hello from browser!");    // popup (browser only)
document.write("Hello DOM");     // writes into the page
Hello, World!
02 Statements & Syntax Rules

A statement is a complete instruction. JavaScript programs are sequences of statements. Each statement ends with a semicolon ; (optional due to ASI — Automatic Semicolon Insertion, but best practice to include).

statements.js
// ── Types of statements ───────────────────────────────────────

// 1. Expression statement
3 + 4;                          // evaluated but result discarded
console.log("expression");     // function call — most common

// 2. Declaration statement
let name = "Alice";             // variable declaration
function greet() {}             // function declaration

// 3. Control flow statement
if (true) { console.log("yes"); }   // conditional
for (let i = 0; i < 3; i++) {}      // loop

// 4. Block statement — groups statements with { }
{
  let x = 10;
  let y = 20;
  console.log(x + y);   // 30
}

// ── Semicolons & whitespace ──────────────────────────────────
let a = 1; let b = 2; let c = a + b;   // multiple on one line (avoid)

// Better: one statement per line
let x = 10;
let y = 20;
let sum = x + y;
console.log(sum);   // 30

// JavaScript is case-sensitive
let Score = 100;
let score = 200;
console.log(Score, score);   // 100  200  — different variables
expression
30
30
100 200
Best practice: Always end statements with ;, use consistent indentation (2 or 4 spaces), and keep one statement per line.

04 Variables — var, let, const

Variables are named containers for storing data values. JavaScript has three keywords for declaring them: var, let, and const.

var — function-scoped (legacy)
var.js
var city = "Buea";
console.log(city);   // "Buea"

var city = "Douala"; // re-declaration allowed (bad practice)
console.log(city);   // "Douala"

city = "Yaounde";    // reassignment allowed
console.log(city);   // "Yaounde"

// var is function-scoped, NOT block-scoped
function test() {
  var x = 5;
  if (true) {
    var x = 10;   // SAME variable — overwrites outer x
    console.log(x); // 10
  }
  console.log(x); // 10  ← leaks out of the if block!
}
test();
"Buea"
"Douala"
"Yaounde"
10
10
let — block-scoped (preferred for mutable variables)
let.js
let score = 0;
console.log(score);   // 0

score = 100;          // reassignment ✅
console.log(score);   // 100

// let score = 200;   // ❌ SyntaxError: cannot re-declare

// let is block-scoped
function test() {
  let y = 5;
  if (true) {
    let y = 10;         // NEW variable inside this block
    console.log(y);     // 10
  }
  console.log(y);       // 5  ← outer y unchanged
}
test();

// let without initial value
let username;
console.log(username);  // undefined
username = "Alice";
console.log(username);  // "Alice"

09 Switch Statement

Best for matching a single value against many possible cases. Uses strict (===) comparison.

switch.js
let day = "Wednesday";

switch (day) {
  case "Monday":
  case "Tuesday":
  case "Wednesday":
  case "Thursday":
  case "Friday":
    console.log("Weekday — class is on!");
    break;                             // ← IMPORTANT: stops fall-through
  case "Saturday":
  case "Sunday":
    console.log("Weekend — no class.");
    break;
  default:
    console.log("Unknown day.");
}
// Weekday — class is on!

// ── Number switch ─────────────────────────────────────────────
let month = 4;
let season;

switch (month) {
  case 12: case 1: case 2:
    season = "Dry season (Harmattan)";
    break;
  case 3: case 4: case 5:
    season = "Early rainy season";
    break;
  case 6: case 7: case 8:
    season = "Mid rainy season";
    break;
  case 9: case 10: case 11:
    season = "Late rainy season";
    break;
  default:
    season = "Invalid month";
}

console.log(`Month ${month}: ${season}`);
// Month 4: Early rainy season

// ── Without break = fall-through (usually a bug) ─────────────
let x = 1;
switch (x) {
  case 1:
    console.log("one");   // runs
    // no break — falls through!

    console.log("two");   // also runs

// ── break — exit the loop immediately ─────────────────────────
console.log("--- break ---");

// Skip specific values
const grades = [85, null, 92, undefined, 78, null, 60];

Function Expression & Arrow Functions ES6
arrow.js
// Function expression — assigned to a variable (NOT hoisted)
const square = function(n) {
  return n * n;
};

// ── Arrow function syntax ─────────────────────────────────────

// Full form
const add = (a, b) => {
  return a + b;
};

// Concise body — implicit return (single expression)
const multiply = (a, b) => a * b;

// Single parameter — no parentheses needed

// No parameters — parentheses required
const getRandom = () => Math.random();

// Return an object — wrap in parentheses

const result = (function() {

function sumAll(...numbers) {

13 Scope & Hoisting
scope.js
// ── Scope ─────────────────────────────────────────────────────
// Global scope — accessible everywhere
let globalVar = "I am global";

function outer() {
  // Function scope
  let outerVar = "I am in outer";

  function inner() {
    // Block scope (also function scope here)
    let innerVar = "I am in inner";

    // Inner can access all outer scopes (scope chain)
    console.log(globalVar);   // ✅ accessible
    console.log(outerVar);    // ✅ accessible
    console.log(innerVar);    // ✅ accessible
  }

  inner();
  // console.log(innerVar); // ❌ ReferenceError — not accessible here
}
outer();

// Block scope with let/const
{
  let blockLet   = "block";
  const blockConst = "also block";
  var blockVar   = "function-scoped";
}
// console.log(blockLet);   // ❌ ReferenceError
console.log(blockVar);       // ✅ "function-scoped" — var leaks!

// ── Hoisting ──────────────────────────────────────────────────
// var declarations are hoisted (initialised to undefined)
console.log(x);     // undefined (not an error)
var x = 5;
console.log(x);     // 5

sayHello();         // ✅ works before definition
function sayHello() {
  console.log("Hello from hoisted function!");

14 Closures

class Student {
  // Private field (ES2022)
  #gpa;

  constructor(name, faculty, gpa) {
    this.name    = name;
    this.faculty = faculty;
    this.#gpa    = gpa;
    this.courses = [];
  }

  // Method
  enroll(course) {
    this.courses.push(course);
    return this;              // return this for chaining
  }

  // Getter
  get info() {
    return `${this.name} | ${this.faculty} | GPA: ${this.#gpa}`;
  }

  // Setter
  set gpa(value) {
    if (value < 0 || value > 4) throw new Error("Invalid GPA");
    this.#gpa = value;
  }

  // Static method — called on the class, not an instance
  static compare(a, b) {
    return b.#gpa - a.#gpa;  // sort descending
  }

  toString() {
    return this.info;
  }
}

// ── Inheritance ────────────────────────────────────────────────
class GradStudent extends Student {
  constructor(name, gpa, thesis) {
    super(name, "Graduate School", gpa);  // call parent constructor
    this.thesis = thesis;
  }

  // Override
  get info() {
    return `${super.info} | Thesis: "${this.thesis}"`;
  }
}

JavaScript is single-threaded. Async operations (API calls, timers, file I/O) are handled via the event loop. Promises and async/await provide clean syntax for async code.

async.js
// ── Promise basics ────────────────────────────────────────────
// States: pending → fulfilled | rejected

function fetchGrade(studentId) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (studentId > 0) {
        resolve({ id: studentId, name: "Alice", grade: 92 });
      } else {
        reject(new Error("Invalid student ID"));
      }
    }, 500);
  });
}

// .then() / .catch() / .finally()
fetchGrade(1)
  .then(data => {
    console.log(`Student: ${data.name}, Grade: ${data.grade}`);
    return data.grade;
  })
  .then(grade => console.log(`Passed: ${grade >= 50}`))
  .catch(err  => console.error("Error:", err.message))
  .finally(()  => console.log("Request finished"));

// Promise combinators
Promise.all([fetchGrade(1), fetchGrade(2), fetchGrade(3)])
  .then(results => results.map(r => `${r.name}:${r.grade}`))
  .then(list    => console.log(list));

// ── async / await ─────────────────────────────────────────────
async function loadReport(id) {
  try {
    const student = await fetchGrade(id);
    console.log(`Loaded: ${student.name}`);

    // Parallel fetch
    const [s1, s2] = await Promise.all([fetchGrade(2), fetchGrade(3)]);
    console.log(`Also loaded: ${s1.name}, ${s2.name}`);

    return {
      main: student,
      others: [s1, s2],
    };
  } catch (err) {
    console.error("Failed:", err.message);
    return null;
  }
}

loadReport(1)
  .then(report => report && console.log("Report ready"));
